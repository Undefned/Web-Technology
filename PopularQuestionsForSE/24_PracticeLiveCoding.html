for (int i = 1; i <= 100; i++)
{
    if (i % 15 == 0) Console.WriteLine("FizzBuzz");
    else if (i % 3 == 0) Console.WriteLine("Fizz");
    else if (i % 5 == 0) Console.WriteLine("Buzz");
    else Console.WriteLine(i);
}

// ✅ С LINQ
Enumerable.Range(1, 100)
    .Select(i => (i % 3 == 0, i % 5 == 0) switch

        (true, false) => "Fizz",
        (false, true) => "Buzz",
        _ => i.ToString()
    })
    .ToList()
    .ForEach(Console.WriteLine);
Проверка строки на палиндром.
обяз

Задача: Проверить, является ли строка палиндромом (читается одинаково слева направо и справа налево).

bool IsPalindrome(string s)
{
    var cleaned = new string(s.Where(char.IsLetterOrDigit).Select(char.ToLower).ToArray());
    int left = 0, right = cleaned.Length - 1;
    while (left < right)
    {
        if (cleaned[left] != cleaned[right]) return false;
        left++;
        right--;
    }
    return true;
}

// ✅ С LINQ
bool IsPalindromeLinq(string s)
{
    var cleaned = s.Where(char.IsLetterOrDigit).Select(char.ToLower).ToArray();
    return cleaned.SequenceEqual(cleaned.Reverse());
}

Задача: Найти все дубликаты в массиве.

// ✅ Через HashSet
int[] FindDuplicates(int[] arr)
{
    var seen = new HashSet<int>();
    var duplicates = new HashSet<int>();
    foreach (int x in arr)
    {
        if (!seen.Add(x))
            duplicates.Add(x);
    }
    return duplicates.ToArray();
}

// ✅ Через LINQ
int[] FindDuplicatesLinq(int[] arr)
{
    return arr
        .GroupBy(x => x)
        .Where(g => g.Count() > 1)
        .Select(g => g.Key)
        .ToArray();
}

Задача: Написать свой метод MyWhere с отложенным выполнением.

public static IEnumerable<T> MyWhere<T>(this IEnumerable<T> source, Func<T, bool> predicate)
{
    if (source == null) throw new ArgumentNullException(nameof(source));
    if (predicate == null) throw new ArgumentNullException(nameof(predicate));
    
    foreach (var item in source)
    {
        if (predicate(item))
            yield return item;
    }
}

// Использование
var numbers = new int[] { 1, 2, 3, 4, 5 };
var result = numbers.MyWhere(x => x > 3);

Написать REST API (CRUD) на ASP.NET Core.

Задача: Написать простой REST API для управления пользователями.

// Model
public class User
{
    public int Id { get; set; }
    public string Name { get; set; }
    public string Email { get; set; }
}

// Controller
[ApiController]
[Route("api/[controller]")]
public class UsersController : ControllerBase
{
    private static List<User> _users = new();
    private static int _nextId = 1;

    [HttpGet]
    public IActionResult GetAll() => Ok(_users);

    [HttpGet("{id}")]
    public IActionResult Get(int id)
    {
        var user = _users.FirstOrDefault(u => u.Id == id);
        return user == null ? NotFound() : Ok(user);
    }

    [HttpPost]
    public IActionResult Create(User user)
    {
        user.Id = _nextId++;
        _users.Add(user);
        return CreatedAtAction(nameof(Get), new { id = user.Id }, user);
    }

    [HttpPut("{id}")]
    public IActionResult Update(int id, User user)
    {
        var existing = _users.FirstOrDefault(u => u.Id == id);
        if (existing == null) return NotFound();
        existing.Name = user.Name;
        existing.Email = user.Email;
        return Ok(existing);
    }

    [HttpDelete("{id}")]
    public IActionResult Delete(int id)
    {
        var user = _users.FirstOrDefault(u => u.Id == id);
        if (user == null) return NotFound();
        _users.Remove(user);
        return NoContent();
    }
}
Написать функцию, проверяющую корректность скобок.

Задача: Проверить, правильно ли расставлены скобки в строке.

bool IsValidParentheses(string s)
{
    var stack = new Stack<char>();
    var pairs = new Dictionary<char, char>
    {
        { ')', '(' },
        { '}', '{' },
        { ']', '[' }
    };

    foreach (char c in s)
    {
        if (pairs.ContainsValue(c)) // Открывающая
            stack.Push(c);
        else if (pairs.ContainsKey(c)) // Закрывающая
        {
            if (stack.Count == 0 || stack.Pop() != pairs[c])
                return false;
        }
    }
    return stack.Count == 0;
}

int FindMissingNumber(int[] arr, int n)
{
    int total = n * (n + 1) / 2;
    int sum = arr.Sum();
    return total - sum;
}

// ✅ С XOR (без переполнения)
int FindMissingNumberXor(int[] arr, int n)
{
    int xor = 0;
    for (int i = 1; i <= n; i++) xor ^= i;
    foreach (int x in arr) xor ^= x;
    return xor;
}
Найти максимальную сумму подмассива (Kadane's algorithm).

Задача: Найти подмассив с максимальной суммой.

int MaxSubArray(int[] arr)
{
    int maxSoFar = arr[0];
    int maxEndingHere = arr[0];
    
    for (int i = 1; i < arr.Length; i++)
    {
        maxEndingHere = Math.Max(arr[i], maxEndingHere + arr[i]);
        maxSoFar = Math.Max(maxSoFar, maxEndingHere);
    }
    return maxSoFar;
}
