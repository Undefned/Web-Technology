🧪 7. Тестирование и Инструменты

Юнит-тесты — `go test` и `*testing.T`

В Go тесты встроены в язык — не нужно устанавливать сторонние фреймворки.

Правила:

  • Файлы тестов заканчиваются на _test.go
  • Функции тестов начинаются с Test
  • Принимают *testing.T

Пример:

// math.go
func Add(a, b int) int {
    return a + b
}

// math_test.go
func TestAdd(t *testing.T) {
    result := Add(2, 3)
    expected := 5
    
    if result != expected {
        t.Errorf("Add(2, 3) = %d; ожидалось %d", result, expected)
    }
}

// Запуск
go test .

Методы testing.T:

  • t.Error(msg) — ошибка (продолжает выполнение)
  • t.Errorf(format, ...) — ошибка с форматированием
  • t.Fatal(msg) — фатальная ошибка (останавливает тест)
  • t.Fatalf(format, ...) — фатальная с форматированием
  • t.Skip(msg) — пропустить тест
  • t.Log(msg) — логирование (показывается с -v)

Запуск с флагами:

go test          # все тесты
go test -v       # подробный вывод
go test -run TestAdd  # только конкретный тест
go test -short   # пропустить долгие тесты (если есть проверка t.Short())
go test -cover   # покрытие кода
Табличные тесты (Table-Driven Tests)

Table-Driven Tests — это паттерн, где тесты описываются в виде таблицы входных данных и ожидаемых результатов.

Зачем:

  • Легко добавлять новые тесты
  • Код не дублируется
  • Хорошо читается
  • Легко видно все случаи

Пример:

func TestAdd(t *testing.T) {
    tests := []struct {
        name     string
        a, b     int
        expected int
    }{
        {"положительные", 2, 3, 5},
        {"отрицательные", -1, -2, -3},
        {"с нулём", 0, 5, 5},
        {"большие числа", 1000000, 2000000, 3000000},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := Add(tt.a, tt.b)
            if result != tt.expected {
                t.Errorf("Add(%d, %d) = %d; ожидалось %d",
                    tt.a, tt.b, result, tt.expected)
            }
        })
    }
}

С тестами на ошибки:

func TestDivide(t *testing.T) {
    tests := []struct {
        name      string
        a, b      float64
        expected  float64
        expectErr bool
    }{
        {"нормальное", 10, 2, 5, false},
        {"деление на ноль", 10, 0, 0, true},
        {"отрицательные", -10, 2, -5, false},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result, err := Divide(tt.a, tt.b)
            
            if tt.expectErr {
                if err == nil {
                    t.Error("ожидалась ошибка, но её нет")
                }
                return
            }
            
            if err != nil {
                t.Errorf("неожиданная ошибка: %v", err)
            }
            
            if result != tt.expected {
                t.Errorf("ожидалось %f, получили %f", tt.expected, result)
            }
        })
    }
}

Преимущества:

  • Добавление нового теста — просто новая строка в таблице
  • Все случаи видны сразу
  • Легко параметризовать
  • Подтесты (t.Run) дают хорошие отчёты
Подтесты — `t.Run()`

t.Run() позволяет создавать вложенные тесты.

Зачем:

  • Группировка связанных тестов
  • Запуск отдельных тестов по имени (-run)
  • Параллельный запуск подтестов
  • Чистые отчёты

Пример:

func TestCalculator(t *testing.T) {
    t.Run("сложение", func(t *testing.T) {
        result := Add(2, 3)
        if result != 5 {
            t.Errorf("Add(2,3) = %d; ожидалось 5", result)
        }
    })
    
    t.Run("вычитание", func(t *testing.T) {
        result := Sub(5, 3)
        if result != 2 {
            t.Errorf("Sub(5,3) = %d; ожидалось 2", result)
        }
    })
}

// Запуск только сложения
go test -run TestCalculator/сложение

Параллельные подтесты:

func TestParallel(t *testing.T) {
    t.Parallel()  // основной тест параллельный
    
    t.Run("тест 1", func(t *testing.T) {
        t.Parallel()
        // тест 1 выполняется параллельно
    })
    
    t.Run("тест 2", func(t *testing.T) {
        t.Parallel()
        // тест 2 выполняется параллельно
    })
}

Вложенные подтесты:

func TestAPI(t *testing.T) {
    t.Run("GET /users", func(t *testing.T) {
        t.Run("существующий пользователь", func(t *testing.T) {
            // ...
        })
        t.Run("несуществующий пользователь", func(t *testing.T) {
            // ...
        })
    })
}
Моки в Go — интерфейсы и ручная реализация

В Go моки обычно делаются через интерфейсы и ручную реализацию.

Шаг 1: Определяем интерфейс для зависимости

type UserRepository interface {
    GetUser(id int) (*User, error)
    SaveUser(user *User) error
}

Шаг 2: Реальная реализация

type PostgresRepo struct {
    db *sql.DB
}

func (r *PostgresRepo) GetUser(id int) (*User, error) {
    // реальная работа с БД
}

Шаг 3: Мок-реализация для тестов

type MockUserRepo struct {
    users map[int]*User
    err   error
}

func (m *MockUserRepo) GetUser(id int) (*User, error) {
    if m.err != nil {
        return nil, m.err
    }
    user, ok := m.users[id]
    if !ok {
        return nil, errors.New("пользователь не найден")
    }
    return user, nil
}

func (m *MockUserRepo) SaveUser(user *User) error {
    if m.err != nil {
        return m.err
    }
    m.users[user.ID] = user
    return nil
}

Шаг 4: Использование в тесте

func TestUserService(t *testing.T) {
    mockRepo := &MockUserRepo{
        users: map[int]*User{
            1: {ID: 1, Name: "Alice"},
        },
    }
    
    service := NewUserService(mockRepo)
    user, err := service.GetUser(1)
    
    if err != nil {
        t.Errorf("ошибка: %v", err)
    }
    if user.Name != "Alice" {
        t.Errorf("ожидалось Alice, получили %s", user.Name)
    }
}

Альтернативы:

  • gomock — генерация моков из интерфейсов
  • testify/mock — ещё один популярный фреймворк
  • httptest — для мока HTTP-серверов

Пример с httptest:

func TestHTTPClient(t *testing.T) {
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        w.Write([]byte(`{"status":"ok"}`))
    }))
    defer server.Close()
    
    // Используем server.URL в тесте
    resp, err := http.Get(server.URL)
    // ... проверяем resp
}
HTTP тестирование — `httptest`

httptest — пакет для тестирования HTTP-серверов и клиентов.

Тестирование обработчиков:

func TestHelloHandler(t *testing.T) {
    // Создаём запрос
    req, err := http.NewRequest("GET", "/hello/Alice", nil)
    if err != nil {
        t.Fatal(err)
    }
    
    // Записываем ответ
    rr := httptest.NewRecorder()
    
    // Вызываем обработчик
    handler := http.HandlerFunc(helloHandler)
    handler.ServeHTTP(rr, req)
    
    // Проверяем статус
    if status := rr.Code; status != http.StatusOK {
        t.Errorf("ожидался статус %v, получили %v", http.StatusOK, status)
    }
    
    // Проверяем тело
    expected := "Hello, Alice!"
    if rr.Body.String() != expected {
        t.Errorf("ожидалось %q, получили %q", expected, rr.Body.String())
    }
}

Тестирование с JSON:

func TestUserHandler(t *testing.T) {
    req, _ := http.NewRequest("GET", "/user/1", nil)
    rr := httptest.NewRecorder()
    
    userHandler.ServeHTTP(rr, req)
    
    var user User
    err := json.NewDecoder(rr.Body).Decode(&user)
    if err != nil {
        t.Fatal(err)
    }
    
    if user.Name != "Alice" {
        t.Errorf("ожидалось Alice, получили %s", user.Name)
    }
}

Тестирование с middleware:

func TestAuthMiddleware(t *testing.T) {
    req, _ := http.NewRequest("GET", "/protected", nil)
    req.Header.Set("Authorization", "valid-token")
    
    rr := httptest.NewRecorder()
    
    // Оборачиваем обработчик в middleware
    handler := authMiddleware(http.HandlerFunc(protectedHandler))
    handler.ServeHTTP(rr, req)
    
    if rr.Code != http.StatusOK {
        t.Errorf("ожидался 200, получили %d", rr.Code)
    }
}

Мок-сервер для клиентов:

func TestExternalAPI(t *testing.T) {
    // Создаём тестовый сервер
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")
        w.Write([]byte(`{"status":"ok"}`))
    }))
    defer server.Close()
    
    // Используем server.URL в клиенте
    resp, err := http.Get(server.URL + "/api/data")
    // ... проверяем ответ
}
Примеры (Examples) — как документация

Примеры (Examples) — это исполняемая документация, которая проверяется тестами.

Правила:

  • Функция начинается с Example
  • Находится в _test.go файлах
  • Вывод проверяется автоматически
  • Показывается в документации go doc

Пример:

// math.go
func Add(a, b int) int {
    return a + b
}

// math_test.go
func ExampleAdd() {
    result := Add(2, 3)
    fmt.Println(result)
    // Output: 5
}

Несколько примеров:

func ExampleAdd_negative() {
    result := Add(-1, -2)
    fmt.Println(result)
    // Output: -3
}

func ExampleAdd_zero() {
    result := Add(0, 5)
    fmt.Println(result)
    // Output: 5
}

Пример для пакета:

func Example() {
    fmt.Println("Пакет mathutils")
    // Output: Пакет mathutils
}

Где используются:

  • В официальной документации Go
  • В pkg.go.dev
  • Проверяются при go test
  • Служат как живые примеры использования

Важно: Если вывод не совпадает с // Output:, тест упадёт.

Test Coverage — покрытие кода

Покрытие кода показывает, какая часть кода выполняется во время тестов.

Зачем:

  • Находит непроверенный код
  • Помогает понять, где нужны дополнительные тесты
  • Метрика качества

Команды:

# Простое покрытие
go test -cover

# Подробный отчёт
go test -coverprofile=coverage.out
go tool cover -html=coverage.out  # открыть в браузере

# Показать в терминале
go tool cover -func=coverage.out

Пример отчёта:

go test -coverprofile=coverage.out
go tool cover -func=coverage.out

math.go:5: Add           100.0%
math.go:10: Sub          100.0%
math.go:15: Divide       80.0%
math.go:22: Multiply     0.0%
total:                   70.0%

Настройка порога:

  • В CI/CD можно требовать минимум 80% покрытия
  • Но 100% покрытие не всегда нужно — фокусируйтесь на важном коде

Флаги:

# Только пакеты с тестами
go test -cover ./...

# Игнорировать некоторые файлы
go test -cover -coverpkg=./... -coverprofile=coverage.out
Инструменты для тестирования — testify, gomega

Сторонние библиотеки упрощают написание тестов.

testify — самый популярный фреймворк для тестов в Go:

import (
    "testing"
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/require"
    "github.com/stretchr/testify/suite"
)

Assert — проверки без остановки:

func TestWithAssert(t *testing.T) {
    result := Add(2, 3)
    
    assert.Equal(t, 5, result, "сложение работает")
    assert.NotNil(t, result)
    assert.Greater(t, result, 0)
    assert.Contains(t, "hello world", "world")
}

Require — проверки с остановкой при ошибке:

func TestWithRequire(t *testing.T) {
    user, err := GetUser(1)
    
    require.NoError(t, err)  // если ошибка — тест останавливается
    require.NotNil(t, user)
    
    assert.Equal(t, "Alice", user.Name)
}

Suite — группировка тестов (как классы в xUnit):

type MySuite struct {
    suite.Suite
    client *Client
}

func (s *MySuite) SetupTest() {
    s.client = NewClient()
}

func (s *MySuite) TearDownTest() {
    s.client.Close()
}

func (s *MySuite) TestAdd() {
    result := s.client.Add(2, 3)
    s.Equal(5, result)
}

func TestSuite(t *testing.T) {
    suite.Run(t, new(MySuite))
}

gomega — BDD-стиль (как в RSpec):

import . "github.com/onsi/gomega"

func TestWithGomega(t *testing.T) {
    RegisterTestingT(t)
    
    result := Add(2, 3)
    
    Ω(result).Should(Equal(5))
    Ω(result).ShouldNot(BeZero())
    Ω(result).Should(BeNumerically(">", 0))
}

Что выбрать:

  • testify — стандарт, много проектов используют
  • gomega — если нравится BDD-стиль
  • Стандартный testing — тоже нормально, особенно для небольших проектов
