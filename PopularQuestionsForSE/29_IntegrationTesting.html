WebApplicationFactory — интеграционное тестирование API.

public class ApiTests : IClassFixture<WebApplicationFactory<Program>>
{
    private readonly HttpClient _client;

    public ApiTests(WebApplicationFactory<Program> factory)
    {
        _client = factory.CreateClient();
    }

    [Fact]
    public async Task GetUsers_ReturnsSuccess()
    {
        var response = await _client.GetAsync("/api/users");
        response.EnsureSuccessStatusCode();
        
        var json = await response.Content.ReadAsStringAsync();
        var users = JsonSerializer.Deserialize<List<User>>(json);
        Assert.NotNull(users);
    }

    [Fact]
    public async Task CreateUser_ReturnsCreated()
    {
        var user = new { Name = "Alice", Email = "alice@example.com" };
        var content = new StringContent(JsonSerializer.Serialize(user), 
            Encoding.UTF8, "application/json");
        
        var response = await _client.PostAsync("/api/users", content);
        Assert.Equal(HttpStatusCode.Created, response.StatusCode);
    }
}

TestContainers — запуск Docker-контейнеров для тестов.

public class DbTests : IClassFixture<DbFixture>
{
    private readonly DbFixture _fixture;

    public DbTests(DbFixture fixture) => _fixture = fixture;

    [Fact]
    public async Task SaveUser_ShouldWork()
    {
        await using var context = _fixture.CreateContext();
        context.Users.Add(new User { Name = "Alice" });
        await context.SaveChangesAsync();
        
        var user = await context.Users.FirstOrDefaultAsync(u => u.Name == "Alice");
        Assert.NotNull(user);
    }
}

public class DbFixture : IAsyncLifetime
{
    private readonly TestcontainerDatabase _container;

    public DbFixture()
    {
        _container = new TestcontainersBuilder<PostgreSqlTestcontainer>()
            .WithDatabase(new PostgreSqlTestcontainerConfiguration
            {
                Database = "testdb",
                Username = "test",
                Password = "test123"
            })
            .Build();
    }

    public async Task InitializeAsync() => await _container.StartAsync();
    public async Task DisposeAsync() => await _container.StopAsync();

    public AppDbContext CreateContext()
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseNpgsql(_container.ConnectionString)
            .Options;
        return new AppDbContext(options);
    }
}
Moq / NSubstitute — мокирование зависимостей.
mid

Moq:

var mock = new Mock<IUserRepository>();
mock.Setup(repo => repo.GetUserAsync(1))
    .ReturnsAsync(new User { Id = 1, Name = "Alice" });

var service = new UserService(mock.Object);
var user = await service.GetUserAsync(1);
Assert.Equal("Alice", user.Name);

// Проверка вызовов
mock.Verify(repo => repo.GetUserAsync(1), Times.Once);

NSubstitute:

var mock = Substitute.For<IUserRepository>();
mock.GetUserAsync(1).Returns(new User { Id = 1, Name = "Alice" });

var service = new UserService(mock);
var user = await service.GetUserAsync(1);
Assert.Equal("Alice", user.Name);

mock.Received(1).GetUserAsync(1);

Тестирование с БД — In-Memory vs реальная БД.
mid

In-Memory:

  • ✅ Быстро
  • ✅ Просто
  • ❌ Не поддерживает транзакции
  • ❌ Не поддерживает некоторые SQL-функции
  • ❌ Может давать ложные результаты
var options = new DbContextOptionsBuilder<AppDbContext>()
    .UseInMemoryDatabase("TestDb")
    .Options;
var context = new AppDbContext(options);

Реальная БД (TestContainers):

  • ✅ Реальное поведение
  • ✅ Все SQL-функции
  • ✅ Транзакции
  • ❌ Медленнее
  • ❌ Требует Docker

Рекомендация: Используйте In-Memory для простых тестов, реальную БД для интеграционных.
