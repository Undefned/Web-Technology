📐 12. SOLID. Архитектурные паттерны Mid+

S — Single Responsibility (единственная ответственность).
обяз

SRP: У класса должна быть только одна причина для изменения.

Пример нарушения:

// ❌ Нарушение — класс делает всё
public class UserService
{
    public void CreateUser(User user) { /* БД */ }
    public void SendEmail(string email) { /* Email */ }
    public void LogActivity(string message) { /* Лог */ }
    public void ValidateUser(User user) { /* Валидация */ }
}

Исправление:

// ✅ Каждый класс отвечает за одну вещь
public class UserService
{
    private readonly IUserRepository _repository;
    private readonly IEmailService _emailService;
    private readonly ILogger _logger;
    private readonly IUserValidator _validator;
    
    public UserService(IUserRepository repository, IEmailService emailService, 
                       ILogger logger, IUserValidator validator)
    {
        _repository = repository;
        _emailService = emailService;
        _logger = logger;
        _validator = validator;
    }
    
    public void CreateUser(User user)
    {
        _validator.Validate(user);
        _repository.Add(user);
        _emailService.SendWelcome(user.Email);
        _logger.Log($"User {user.Id} created");
    }
}
O — Open/Closed (открыт для расширения, закрыт для изменения).
обяз

OCP: Классы открыты для расширения, но закрыты для модификации.

Пример нарушения:

// ❌ Нарушение — нужно менять при добавлении нового типа
public class DiscountCalculator
{
    public decimal Calculate(decimal price, string type)
    {
        if (type == "Regular") return price * 0.9m;
        if (type == "VIP") return price * 0.8m;
        if (type == "Employee") return price * 0.7m;
        return price;
    }
}

Исправление:

// ✅ Открыт для расширения
public interface IDiscountStrategy
{
    decimal Apply(decimal price);
}

public class RegularDiscount : IDiscountStrategy
{
    public decimal Apply(decimal price) => price * 0.9m;
}

public class VIPDiscount : IDiscountStrategy
{
    public decimal Apply(decimal price) => price * 0.8m;
}

public class DiscountCalculator
{
    public decimal Calculate(decimal price, IDiscountStrategy strategy)
    {
        return strategy.Apply(price);
    }
}

// Добавление нового типа без изменения существующего кода
public class HolidayDiscount : IDiscountStrategy
{
    public decimal Apply(decimal price) => price * 0.5m;
}
L — Liskov Substitution (подстановка Барбары Лисков).
обяз

LSP: Объекты производного класса должны заменять объекты базового класса без изменения поведения.

Пример нарушения:

// ❌ Нарушение — производный класс не может заменить базовый
public class Rectangle
{
    public int Width { get; set; }
    public int Height { get; set; }
    
    public int Area() => Width * Height;
}

public class Square : Rectangle
{
    public new int Width 
    {
        set { base.Width = base.Height = value; }
    }
    
    public new int Height
    {
        set { base.Width = base.Height = value; }
    }
}

// Ожидание сломается
void Test(Rectangle rect)
{
    rect.Width = 4;
    rect.Height = 5;
    Console.WriteLine(rect.Area());  // Ожидаем 20, но Square даст 25
}

Исправление:

// ✅ Правильное проектирование
public abstract class Shape
{
    public abstract int Area();
}

public class Rectangle : Shape
{
    public int Width { get; set; }
    public int Height { get; set; }
    
    public override int Area() => Width * Height;
}

public class Square : Shape
{
    public int Side { get; set; }
    public override int Area() => Side * Side;
}
I — Interface Segregation (разделение интерфейсов).
обяз

ISP: Интерфейсы должны быть маленькими и специфичными.

Пример нарушения:

// ❌ Нарушение — интерфейс слишком большой
public interface IEmployee
{
    void Work();
    void Eat();
    void Sleep();
    void Code();
    void Manage();
    void Test();
    void Deploy();
}

public class Developer : IEmployee
{
    public void Work() { }
    public void Eat() { }
    public void Sleep() { }
    public void Code() { }
    public void Manage() { throw new NotImplementedException(); }  // ❌
    public void Test() { }
    public void Deploy() { }
}

Исправление:

// ✅ Разделение интерфейсов
public interface IWorker { void Work(); }
public interface IDeveloper { void Code(); void Test(); }
public interface IManager { void Manage(); }
public interface IDeployer { void Deploy(); }

public class Developer : IWorker, IDeveloper, IDeployer
{
    public void Work() { Code(); Test(); Deploy(); }
    public void Code() { }
    public void Test() { }
    public void Deploy() { }
}
D — Dependency Inversion (инверсия зависимостей).
обяз

DIP: Зависимости должны быть от абстракций, а не от конкретных реализаций.

Пример нарушения:

// ❌ Нарушение — зависит от конкретной реализации
public class UserService
{
    private SqlUserRepository _repository = new SqlUserRepository();  // Жёсткая связь
    
    public void Save(User user)
    {
        _repository.Save(user);
    }
}

Исправление:

// ✅ Инверсия зависимостей
public interface IUserRepository
{
    void Save(User user);
}

public class SqlUserRepository : IUserRepository
{
    public void Save(User user) { /* SQL */ }
}

public class UserService
{
    private readonly IUserRepository _repository;
    
    public UserService(IUserRepository repository)
    {
        _repository = repository;
    }
    
    public void Save(User user)
    {
        _repository.Save(user);
    }
}

// Тестирование с моком
var mock = new Mock<IUserRepository>();
var service = new UserService(mock.Object);
IoC (Inversion of Control) — что это, зачем.
обяз

IoC (Инверсия управления) — принцип, при котором управление передаётся внешнему контейнеру.

Проблема:

// ❌ Без IoC — класс создаёт свои зависимости
public class OrderService
{
    private EmailService _email = new EmailService();  // Жёсткая связь
    private Logger _logger = new Logger();
    
    public void ProcessOrder(Order order)
    {
        _logger.Log("Processing");
        // ...
        _email.Send("Order processed");
    }
}

Решение (IoC):

// ✅ С IoC — зависимости передаются извне
public class OrderService
{
    private readonly IEmailService _email;
    private readonly ILogger _logger;
    
    public OrderService(IEmailService email, ILogger logger)
    {
        _email = email;
        _logger = logger;
    }
    
    public void ProcessOrder(Order order)
    {
        _logger.Log("Processing");
        // ...
        _email.Send("Order processed");
    }
}

Преимущества:

  • Слабая связанность
  • Легкое тестирование (моки)
  • Гибкость (легко заменить реализации)
  • Упрощение поддержки кода
DI (Dependency Injection) — конструкторная, свойств, методовая.
обяз

1. Constructor Injection (рекомендуется)

  • Зависимости передаются через конструктор
  • Объект всегда в валидном состоянии
  • Легко тестировать
public class OrderService
{
    private readonly IEmailService _email;
    private readonly ILogger _logger;
    
    public OrderService(IEmailService email, ILogger logger)
    {
        _email = email;
        _logger = logger;
    }
}

2. Property Injection (для опциональных зависимостей)

  • Зависимости через публичные свойства
  • Для опциональных или второстепенных зависимостей
public class OrderService
{
    public ILogger Logger { get; set; }  // Необязательная зависимость
    
    public void Process()
    {
        Logger?.Log("Processing");
    }
}

3. Method Injection (для временных зависимостей)

  • Зависимости передаются в метод
  • Для параметров, которые меняются каждый вызов
public class OrderService
{
    public void Process(Order order, IEmailService email)
    {
        email.Send("Order " + order.Id);
    }
}

Сравнение:

Тип Когда использовать Сложность
Constructor Всегда для обязательных зависимостей Низкая
Property Опциональные зависимости Средняя
Method Временные или изменяющиеся зависимости Низкая
Cohesion (связность) vs Coupling (сцепление) — разница.
mid

Cohesion (Связность) — насколько хорошо элементы внутри модуля связаны друг с другом.

  • Высокая связность: все элементы модуля работают вместе для одной цели (хорошо)
  • Низкая связность: элементы модуля делают разные, несвязанные вещи (плохо)

Coupling (Сцепление) — насколько модули зависят друг от друга.

  • Низкое сцепление: модули слабо зависят друг от друга (хорошо)
  • Высокое сцепление: модули сильно зависят друг от друга (плохо)

Примеры:

// ❌ Низкая связность (модуль делает всё)
public class Utility
{
    public void CalculateDiscount() { }
    public void SendEmail() { }
    public void LogError() { }
}

// ✅ Высокая связность (модуль делает одно)
public class DiscountCalculator
{
    public decimal Calculate(decimal price) { }
}

Цель:

  • Высокая связность внутри модулей
  • Низкое сцепление между модулями
  • Легче поддерживать, тестировать, изменять
Паттерны GoF: Singleton, Factory, Abstract Factory, Builder.
обяз

Singleton (Одиночка)

  • Гарантирует один экземпляр класса
  • Глобальная точка доступа
  • Антипаттерн для тестирования
public sealed class Singleton
{
    private static readonly Singleton _instance = new();
    private Singleton() { }
    public static Singleton Instance => _instance;
}

Factory Method (Фабричный метод)

  • Создаёт объект, позволяя наследникам выбирать тип
public abstract class Creator
{
    public abstract IProduct CreateProduct();
}

Abstract Factory (Абстрактная фабрика)

  • Создаёт семейства связанных объектов
public interface IUIFactory
{
    IButton CreateButton();
    IWindow CreateWindow();
}

Builder (Строитель)

  • Пошаговое создание сложных объектов
var builder = new FluentBuilder();
builder
    .SetName("Product")
    .SetPrice(100)
    .SetCategory("Electronics")
    .Build();
Паттерны: Strategy, Observer, Decorator, Adapter, Facade.
mid

Strategy (Стратегия)

  • Выбор алгоритма во время выполнения
public interface ISortStrategy { void Sort(List<int> list); }
public class QuickSort : ISortStrategy { ... }
public class MergeSort : ISortStrategy { ... }

Observer (Наблюдатель)

  • Уведомление об изменениях (события)
public class Publisher
{
    public event EventHandler OnChange;
    public void Change() => OnChange?.Invoke(this, EventArgs.Empty);
}

Decorator (Декоратор)

  • Добавление функциональности без изменения класса
public interface IComponent { void Operation(); }
public class Decorator : IComponent
{
    private readonly IComponent _component;
    public Decorator(IComponent comp) => _component = comp;
    public void Operation() { _component.Operation(); /* Доп. логика */ }
}

Adapter (Адаптер)

  • Адаптация интерфейсов разных систем
public class LegacyService { public string GetData() => "Legacy"; }
public class Adapter : INewService
{
    private readonly LegacyService _legacy = new();
    public string GetNewData() => _legacy.GetData();
}

Facade (Фасад)

  • Упрощённый интерфейс для сложной подсистемы
public class Facade
{
    public void DoWork()
    {
        new SubSystem1().Do();
        new SubSystem2().Do();
        new SubSystem3().Do();
    }
}
Паттерны: Repository, Unit of Work.
обяз

Repository (Репозиторий)

  • Абстракция над хранилищем данных
  • CRUD операции для сущностей
public interface IRepository<T> where T : class
{
    Task<T> GetByIdAsync(int id);
    Task<IEnumerable<T>> GetAllAsync();
    Task AddAsync(T entity);
    void Update(T entity);
    void Delete(T entity);
}

public class UserRepository : IRepository<User>
{
    private readonly AppDbContext _context;
    public UserRepository(AppDbContext context) => _context = context;
    
    public async Task<User> GetByIdAsync(int id)
        => await _context.Users.FindAsync(id);
}

Unit of Work (Единица работы)

  • Группировка операций в транзакцию
  • Координация нескольких репозиториев
public interface IUnitOfWork : IDisposable
{
    IUserRepository Users { get; }
    IOrderRepository Orders { get; }
    Task<int> SaveChangesAsync();
}

public class UnitOfWork : IUnitOfWork
{
    private readonly AppDbContext _context;
    private IUserRepository _users;
    private IOrderRepository _orders;
    
    public IUserRepository Users 
        => _users ??= new UserRepository(_context);
    
    public IOrderRepository Orders 
        => _orders ??= new OrderRepository(_context);
    
    public async Task<int> SaveChangesAsync()
        => await _context.SaveChangesAsync();
}

// Использование
using var uow = new UnitOfWork();
var user = await uow.Users.GetByIdAsync(1);
var order = new Order { UserId = user.Id };
await uow.Orders.AddAsync(order);
await uow.SaveChangesAsync();  // Одна транзакция
Clean Architecture / Onion Architecture — базово.
mid

Clean Architecture — архитектура с чёткими слоями и зависимостями, направленными к центру.

Слои (от внешнего к внутреннему):

  • Presentation (UI): контроллеры, UI
  • Application: бизнес-логика, DTO, валидация
  • Domain: сущности, интерфейсы репозиториев
  • Infrastructure: БД, внешние сервисы, реализация репозиториев

Структура:

// Domain Layer
public class User
{
    public int Id { get; set; }
    public string Name { get; set; }
}

public interface IUserRepository
{
    Task<User> GetByIdAsync(int id);
}

// Application Layer
public class UserService
{
    private readonly IUserRepository _repository;
    public UserService(IUserRepository repository) => _repository = repository;
    
    public async Task<UserDto> GetUserAsync(int id)
    {
        var user = await _repository.GetByIdAsync(id);
        return new UserDto { Id = user.Id, Name = user.Name };
    }
}

// Infrastructure Layer
public class SqlUserRepository : IUserRepository
{
    private readonly AppDbContext _context;
    public SqlUserRepository(AppDbContext context) => _context = context;
    
    public async Task<User> GetByIdAsync(int id)
        => await _context.Users.FindAsync(id);
}

// Presentation Layer (ASP.NET Core)
[ApiController]
[Route("api/users")]
public class UsersController : ControllerBase
{
    private readonly UserService _service;
    public UsersController(UserService service) => _service = service;
    
    [HttpGet("{id}")]
    public async Task<IActionResult> Get(int id)
        => Ok(await _service.GetUserAsync(id));
}

Преимущества:

  • Независимость от внешних фреймворков
  • Легко тестировать бизнес-логику
  • Гибкость (легко менять БД или UI)
  • Чёткое разделение ответственности
Микросервисы vs Монолит — плюсы/минусы, когда выбирать.
обяз

Монолит:

  • Всё приложение в одной сборке
  • Одна БД, один деплой
  • Простая разработка на старте
  • Сложное масштабирование
  • Одна технология

Микросервисы:

  • Много маленьких сервисов
  • Каждый сервис — отдельная БД
  • Сложная разработка
  • Гибкое масштабирование
  • Разные технологии
Характеристика Монолит Микросервисы
Разработка Простая Сложная
Масштабирование Сложное Простое
Деплой Один Много
Команда Маленькая Большая
Производительность Лучше (локальные вызовы) Хуже (сетевые вызовы)
Технологии Одна Разные

Когда выбирать:

  • Монолит: стартап, маленькая команда, простые требования
  • Микросервисы: большая команда, высокие нагрузки, сложная бизнес-логика
DDD (Domain-Driven Design) — Entity, Value Object, Aggregate.
mid

Entity (Сущность)

  • Имеет идентичность (Id)
  • Изменяема
  • Пример: User, Order
public class User
{
    public int Id { get; private set; }  // Идентичность
    public string Name { get; set; }
    public string Email { get; set; }
}

Value Object (Объект-значение)

  • Нет идентичности
  • Неизменяемый (immutable)
  • Сравнивается по значению
  • Пример: Address, Money, Color
public record Address
{
    public string Street { get; init; }
    public string City { get; init; }
    public string ZipCode { get; init; }
}

public record Money
{
    public decimal Amount { get; init; }
    public string Currency { get; init; }
    
    public Money Add(Money other)
    {
        if (Currency != other.Currency)
            throw new Exception("Different currencies");
        return new Money { Amount = Amount + other.Amount, Currency = Currency };
    }
}

Aggregate (Агрегат)

  • Группа связанных сущностей
  • Одна корневая сущность (Aggregate Root)
  • Транзакционная граница
  • Пример: Order (корень) + OrderItems
public class Order  // Aggregate Root
{
    private List<OrderItem> _items = new();
    public int Id { get; private set; }
    public Money Total { get; private set; }
    public IReadOnlyList<OrderItem> Items => _items;
    
    public void AddItem(string productName, Money price, int quantity)
    {
        _items.Add(new OrderItem(productName, price, quantity));
        RecalculateTotal();
    }
    
    private void RecalculateTotal()
    {
        Total = _items.Aggregate(new Money(), (sum, item) => sum.Add(item.Price));
    }
}

public class OrderItem  // Внутри агрегата
{
    public string ProductName { get; }
    public Money Price { get; }
    public int Quantity { get; }
    
    public OrderItem(string productName, Money price, int quantity)
    {
        ProductName = productName;
        Price = price;
        Quantity = quantity;
    }
}
CQRS — Command Query Responsibility Segregation.
mid

CQRS — разделение операций чтения и записи.

Принцип:

  • Command — изменения (CREATE, UPDATE, DELETE)
  • Query — чтение (SELECT)
  • Разные модели для чтения и записи

Пример:

// Command — изменение
public class CreateOrderCommand
{
    public int UserId { get; set; }
    public List<OrderItemDto> Items { get; set; }
}

public class CreateOrderCommandHandler
{
    public async Task Handle(CreateOrderCommand command)
    {
        // Создание заказа в БД
    }
}

// Query — чтение
public class GetOrderQuery
{
    public int OrderId { get; set; }
}

public class GetOrderQueryHandler
{
    public async Task<OrderDto> Handle(GetOrderQuery query)
    {
        // Чтение из отдельной read-модели (или из кэша)
    }
}

// Использование
await mediator.Send(new CreateOrderCommand { ... });
var order = await mediator.Send(new GetOrderQuery { OrderId = 1 });

Когда использовать:

  • Сложная бизнес-логика
  • Разные требования к чтению и записи
  • Высокая нагрузка (отдельное масштабирование)
  • Event Sourcing + CQRS (храним события, читаем проекции)

Сложности:

  • Усложнение кода
  • Консистентность (eventual consistency)
  • Больше кода
