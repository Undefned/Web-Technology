🏗️ 8. Типичные паттерны в Go и Go для микросервисов

Функциональные опции (Functional Options) — паттерн

Functional Options — это паттерн для гибкой настройки объектов с множеством параметров.

Проблема:

  • У структуры много полей с настройками
  • Часть из них опциональна
  • Не хочется создавать десятки конструкторов

Решение — Functional Options:

type Server struct {
    host    string
    port    int
    timeout time.Duration
    maxConn int
}

// Тип опции — функция, которая меняет Server
type Option func(*Server)

// Опции-конструкторы
func WithPort(port int) Option {
    return func(s *Server) {
        s.port = port
    }
}

func WithTimeout(timeout time.Duration) Option {
    return func(s *Server) {
        s.timeout = timeout
    }
}

func WithMaxConn(maxConn int) Option {
    return func(s *Server) {
        s.maxConn = maxConn
    }
}

// Конструктор с опциями
func NewServer(host string, opts ...Option) *Server {
    // Значения по умолчанию
    s := &Server{
        host:    host,
        port:    8080,
        timeout: 30 * time.Second,
        maxConn: 100,
    }
    
    // Применяем опции
    for _, opt := range opts {
        opt(s)
    }
    return s
}

Использование:

// С настройками по умолчанию
s1 := NewServer("localhost")

// С кастомными настройками
s2 := NewServer("localhost",
    WithPort(9090),
    WithTimeout(10*time.Second),
    WithMaxConn(200),
)

Преимущества:

  • Гибкость — можно передать любые опции
  • Читаемость — понятно, что настраивается
  • Расширяемость — легко добавить новые опции
  • Безопасность — не нужно передавать nil

Где используется:

  • gRPC (grpc.Dial с опциями)
  • HTTP-серверы
  • Конфигурация клиентов
  • Любые объекты с настройками
Веб-фреймворки — почему часто хватает `net/http`?

Стандартный net/http в Go настолько хорош, что многие проекты обходятся без фреймворков.

Что есть в net/http:

  • Роутинг (с Go 1.22 — улучшенный паттерн-матчинг)
  • Middleware (через обёртки)
  • Работа с сессиями/куками
  • Шаблонизация
  • Грамотный HTTP-клиент

Пример без фреймворка:

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("GET /users/{id}", handleUser)
    mux.HandleFunc("POST /users", createUser)
    
    handler := loggingMiddleware(mux)
    handler = authMiddleware(handler)
    
    http.ListenAndServe(":8080", handler)
}

Когда нужен фреймворк:

  • Gin — если нужна максимальная производительность
  • Echo — если нужна производительность + удобный роутинг
  • Fiber — если нужна производительность как у Express.js
  • Beego/Revel — если нужен полноценный MVC-фреймворк

Сравнение производительности (условное):

  • net/http — быстрый, но чуть медленнее специализированных
  • Gin — быстрее net/http на ~20-30%
  • Fiber — ещё быстрее, но использует fasthttp
  • Для большинства проектов net/http достаточно

Совет: Начните с net/http. Если станет не хватать — мигрируйте на фреймворк.

Middleware — концепция и пример

Middleware (промежуточное ПО) — это функции, которые оборачивают обработчики и выполняются до или после них.

Типичные задачи:

  • Логирование
  • Аутентификация
  • Сжатие ответов
  • Ограничение скорости (rate limiting)
  • Восстановление после паники
  • Добавление заголовков

Пример middleware для логирования:

func loggingMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        
        // Создаём обёртку для записи статуса
        wrapped := &responseWriter{ResponseWriter: w}
        
        // Вызываем следующий обработчик
        next.ServeHTTP(wrapped, r)
        
        // Логируем после
        log.Printf("%s %s %d %v",
            r.Method,
            r.URL.Path,
            wrapped.status,
            time.Since(start),
        )
    })
}

// Обёртка для захвата статуса ответа
type responseWriter struct {
    http.ResponseWriter
    status int
}

func (rw *responseWriter) WriteHeader(code int) {
    rw.status = code
    rw.ResponseWriter.WriteHeader(code)
}

Цепочка middleware:

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("/", mainHandler)
    
    // Оборачиваем в обратном порядке
    handler := mux
    handler = loggingMiddleware(handler)
    handler = authMiddleware(handler)
    handler = recoveryMiddleware(handler)
    
    http.ListenAndServe(":8080", handler)
}

Короткая запись:

type Middleware func(http.Handler) http.Handler

func Chain(h http.Handler, middlewares ...Middleware) http.Handler {
    for _, m := range middlewares {
        h = m(h)
    }
    return h
}

// Использование
handler := Chain(mux, loggingMiddleware, authMiddleware, recoveryMiddleware)
Базы данных — `database/sql` vs ORM

В Go есть два основных подхода к работе с БД: чистый SQL и ORM.

database/sql — стандартный пакет для работы с SQL:

import "database/sql"
import _ "github.com/lib/pq"  // драйвер PostgreSQL

func main() {
    db, err := sql.Open("postgres", "postgres://user:pass@localhost/db")
    if err != nil {
        panic(err)
    }
    defer db.Close()
    
    // Запрос
    rows, err := db.Query("SELECT id, name FROM users WHERE age > $1", 18)
    if err != nil {
        // обработка
    }
    defer rows.Close()
    
    var users []User
    for rows.Next() {
        var u User
        err := rows.Scan(&u.ID, &u.Name)
        // ...
    }
}

ORM — GORM (самый популярный):

import "gorm.io/gorm"

type User struct {
    ID   uint
    Name string
    Age  int
}

func main() {
    db, err := gorm.Open(postgres.Open(dsn), &gorm.Config{})
    
    // Автомиграция
    db.AutoMigrate(&User{})
    
    // CRUD
    db.Create(&User{Name: "Alice", Age: 30})
    
    var user User
    db.First(&user, 1)  // по ID
    
    db.Where("age > ?", 18).Find(&users)
}

Когда что использовать:

ПодходПлюсыМинусы
database/sql Полный контроль, быстрее, понятный SQL Много кода, сложно с JOIN
ORM (GORM, ent) Меньше кода, автоматическая миграция, удобно Медленнее, сложные запросы сложно написать

Компромисс — Query Builder:

  • sqlx — расширение database/sql (удобное сканирование)
  • squirrel — построитель запросов
  • sqlc — генерирует Go-код из SQL
Почему Go хорош для микросервисов?

Go стал одним из главных языков для микросервисов. Причины:

1. Маленький бинарник

  • Собранная программа — один файл без зависимостей
  • Размер ~10-30 МБ (в зависимости от проекта)
  • Быстрое развёртывание

2. Быстрый старт

  • Нет виртуальной машины (как JVM или .NET Runtime)
  • Запускается за миллисекунды
  • Идеально для авто-скейлинга (Kubernetes)

3. Низкое потребление памяти

  • Горутины легковесны (2 КБ стек)
  • Можно запускать тысячи микросервисов на одном сервере

4. Встроенная конкурентность

  • Легко обрабатывать много одновременных запросов
  • Каналы и горутины упрощают асинхронную работу

5. Отличная стандартная библиотека

  • HTTP-сервер, gRPC, шифрование, JSON — всё уже есть
  • Минимум внешних зависимостей

6. Простота

  • Код легко читать и поддерживать
  • Строгий синтаксис и форматирование
  • Меньше магии → меньше багов

7. Кросскомпиляция

  • Одна команда: GOOS=linux GOARCH=amd64 go build
  • Собираем под Linux с Windows/Mac
gRPC в Go — `protobuf` и генерация кода

gRPC — это высокопроизводительный RPC-фреймворк от Google.

Преимущества перед REST:

  • Быстрее (бинарный протокол)
  • Строгая типизация (protobuf)
  • Поддержка стримов (потоковая передача)
  • Автоматическая генерация клиентов и серверов

Шаг 1: Определяем протокол (.proto файл)

// hello.proto
syntax = "proto3";

service Greeter {
    rpc SayHello (HelloRequest) returns (HelloResponse);
}

message HelloRequest {
    string name = 1;
}

message HelloResponse {
    string message = 1;
}

Шаг 2: Генерируем код

protoc --go_out=. --go-grpc_out=. hello.proto

Шаг 3: Реализуем сервер

type server struct {
    pb.UnimplementedGreeterServer
}

func (s *server) SayHello(ctx context.Context, req *pb.HelloRequest) (*pb.HelloResponse, error) {
    return &pb.HelloResponse{
        Message: "Hello, " + req.Name,
    }, nil
}

func main() {
    lis, _ := net.Listen("tcp", ":50051")
    s := grpc.NewServer()
    pb.RegisterGreeterServer(s, &server{})
    s.Serve(lis)
}

Шаг 4: Клиент

conn, _ := grpc.Dial("localhost:50051", grpc.WithInsecure())
defer conn.Close()

client := pb.NewGreeterClient(conn)
resp, _ := client.SayHello(context.Background(), &pb.HelloRequest{Name: "Alice"})
fmt.Println(resp.Message)  // "Hello, Alice"

Стримы:

// Серверный стрим
rpc GetUsers (Empty) returns (stream User);

// Клиентский стрим
rpc UploadData (stream Data) returns (Status);

// Двунаправленный стрим
rpc Chat (stream Message) returns (stream Message);
Graceful Shutdown — корректное завершение сервера

Graceful Shutdown — это завершение сервера с ожиданием текущих запросов.

Зачем:

  • Не обрывать запросы, которые уже обрабатываются
  • Дать время на завершение операций
  • Не допустить ошибок у клиентов

Реализация:

func main() {
    server := &http.Server{
        Addr:    ":8080",
        Handler: mux,
    }
    
    // Канал для сигналов
    stop := make(chan os.Signal, 1)
    signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
    
    // Запускаем сервер в горутине
    go func() {
        if err := server.ListenAndServe(); err != http.ErrServerClosed {
            log.Fatal(err)
        }
    }()
    
    // Ждём сигнал
    <-stop
    log.Println("Получен сигнал остановки")
    
    // Даём время на завершение текущих запросов
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    if err := server.Shutdown(ctx); err != nil {
        log.Fatal("Ошибка при завершении:", err)
    }
    
    log.Println("Сервер завершён корректно")
}

Для gRPC:

func main() {
    s := grpc.NewServer()
    // ... регистрация
    
    go func() {
        s.Serve(lis)
    }()
    
    <-stop
    s.GracefulStop()  // gRPC имеет встроенный GracefulStop
}
Обработка сигналов — `os/signal`

Пакет os/signal позволяет реагировать на системные сигналы.

Основные сигналы:

  • SIGINT — Ctrl+C
  • SIGTERM — запрос на завершение (от Kubernetes)
  • SIGKILL — принудительное убийство (нельзя перехватить)
  • SIGUSR1 — пользовательский сигнал

Пример:

func main() {
    // Создаём канал для сигналов
    sigChan := make(chan os.Signal, 1)
    
    // Подписываемся на сигналы
    signal.Notify(sigChan, 
        syscall.SIGINT,
        syscall.SIGTERM,
        syscall.SIGUSR1,
    )
    
    go func() {
        for sig := range sigChan {
            switch sig {
            case syscall.SIGINT, syscall.SIGTERM:
                log.Println("Получен сигнал завершения:", sig)
                // graceful shutdown
                os.Exit(0)
            case syscall.SIGUSR1:
                log.Println("Получен SIGUSR1 — перезагружаем конфиг")
                reloadConfig()
            }
        }
    }()
    
    // ... работа программы
}

Игнорирование сигналов:

signal.Ignore(syscall.SIGPIPE)  // игнорируем SIGPIPE

Важно:

  • Канал для сигналов должен быть буферизированным
  • Не используйте os.Exit в горутинах — используйте каналы для оповещения main
Чистая архитектура в Go — слои

Чистая архитектура (Clean Architecture) — способ организации кода с чёткими слоями.

Слои (от внешнего к внутреннему):

  1. Presentation (представление) — HTTP-хендлеры, gRPC, CLI
  2. Use Case (бизнес-логика) — сценарии использования приложения
  3. Domain (сущности) — бизнес-сущности, правила
  4. Infrastructure (инфраструктура) — БД, внешние API, кэш

Структура проекта:

myapp/
├── cmd/
│   └── server/
│       └── main.go
├── internal/
│   ├── domain/          # бизнес-сущности
│   │   ├── user.go
│   │   └── order.go
│   ├── usecase/         # бизнес-логика
│   │   ├── user.go
│   │   └── order.go
│   ├── handler/         # HTTP-хендлеры
│   │   └── user.go
│   └── repository/      # работа с БД
│       ├── user.go
│       └── postgres/
│           └── user.go
└── pkg/                 # общие утилиты

Пример usecase:

type UserUseCase struct {
    repo UserRepository
}

func (uc *UserUseCase) GetUser(id int) (*User, error) {
    user, err := uc.repo.FindByID(id)
    if err != nil {
        return nil, err
    }
    // Бизнес-логика
    if user.Age < 18 {
        user.IsAdult = false
    }
    return user, nil
}

Преимущества:

  • Бизнес-логика не зависит от фреймворков
  • Легко тестировать (можно подменить репозитории)
  • Легко менять инфраструктуру (БД, внешние сервисы)
Context в HTTP — `r.Context()` и передача данных

В HTTP-запросах контекст используется для передачи данных и управления таймаутами.

Основные возможности:

  • Отмена запроса (клиент закрыл соединение)
  • Таймауты
  • Передача метаданных (ID пользователя, ID запроса)

Пример с таймаутом:

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("/api", apiHandler)
    
    // Таймаут для всех запросов
    server := &http.Server{
        Addr:         ":8080",
        Handler:      mux,
        ReadTimeout:  5 * time.Second,
        WriteTimeout: 10 * time.Second,
    }
}

Передача данных через контекст:

// Ключ для контекста (тип — любой, но лучше использовать свой тип)
type contextKey string

const userIDKey contextKey = "userID"

// Middleware — добавляем UserID в контекст
func authMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // Получаем UserID из токена
        userID := getUserIDFromToken(r.Header.Get("Authorization"))
        
        // Добавляем в контекст
        ctx := context.WithValue(r.Context(), userIDKey, userID)
        next.ServeHTTP(w, r.WithContext(ctx))
    })
}

// Хендлер — читаем из контекста
func apiHandler(w http.ResponseWriter, r *http.Request) {
    userID, ok := r.Context().Value(userIDKey).(int)
    if !ok {
        http.Error(w, "Unauthorized", http.StatusUnauthorized)
        return
    }
    fmt.Fprintf(w, "User ID: %d", userID)
}

Реакция на отмену запроса:

func longHandler(w http.ResponseWriter, r *http.Request) {
    select {
    case <-r.Context().Done():
        // Клиент отключился или таймаут
        log.Println("Запрос отменён")
        return
    case <-time.After(10 * time.Second):
        // Долгая операция завершилась
        w.Write([]byte("Done"))
    }
}

Таймауты для обработчиков:

func handlerWithTimeout(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        ctx, cancel := context.WithTimeout(r.Context(), 5*time.Second)
        defer cancel()
        next.ServeHTTP(w, r.WithContext(ctx))
    })
}
