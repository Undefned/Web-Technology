🚀 Gin — Web-фреймворк для Go

Что такое Gin и зачем он нужен?

Gin — это самый популярный веб-фреймворк для Go. Он написан на Go и использует net/http под капотом.

Зачем нужен Gin:

  • Упрощает создание REST API
  • Даёт удобный роутинг (маршрутизацию)
  • Встроенные middleware (логирование, CORS, восстановление после паники)
  • Быстрый — один из самых быстрых фреймворков для Go
  • Поддержка JSON, XML, YAML

Минимальный пример:

package main

import "github.com/gin-gonic/gin"

func main() {
    r := gin.Default()
    
    r.GET("/ping", func(c *gin.Context) {
        c.JSON(200, gin.H{
            "message": "pong",
        })
    })
    
    r.Run() // слушает на :8080
}

Установка:

go get -u github.com/gin-gonic/gin
Роутинг в Gin — основные методы

Gin поддерживает все стандартные HTTP-методы.

func main() {
    r := gin.Default()
    
    // GET
    r.GET("/users", getUsers)
    r.GET("/users/:id", getUserByID)
    
    // POST
    r.POST("/users", createUser)
    
    // PUT
    r.PUT("/users/:id", updateUser)
    
    // PATCH
    r.PATCH("/users/:id", patchUser)
    
    // DELETE
    r.DELETE("/users/:id", deleteUser)
    
    // HEAD, OPTIONS
    r.HEAD("/ping", headHandler)
    r.OPTIONS("/users", optionsHandler)
    
    r.Run()
}

Параметры в URL:

// /users/123 → id = "123"
r.GET("/users/:id", func(c *gin.Context) {
    id := c.Param("id")
    c.String(200, "User ID: %s", id)
})

// /users/123/posts/456 → id = "123", postId = "456"
r.GET("/users/:id/posts/:postId", func(c *gin.Context) {
    id := c.Param("id")
    postId := c.Param("postId")
    c.String(200, "User %s, Post %s", id, postId)
})

// /users?page=2&limit=10 → page=2, limit=10
r.GET("/users", func(c *gin.Context) {
    page := c.DefaultQuery("page", "1")
    limit := c.Query("limit")
    c.String(200, "page=%s, limit=%s", page, limit)
})
Обработка JSON — приём и отправка

Gin имеет встроенную поддержку JSON.

Отправка JSON:

type User struct {
    ID   int    `json:"id"`
    Name string `json:"name"`
    Age  int    `json:"age"`
}

r.GET("/user", func(c *gin.Context) {
    user := User{ID: 1, Name: "Alice", Age: 30}
    c.JSON(200, user)
})

// Ответ: {"id":1,"name":"Alice","age":30}

Приём JSON:

r.POST("/user", func(c *gin.Context) {
    var user User
    
    // Привязываем JSON к структуре
    if err := c.ShouldBindJSON(&user); err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }
    
    // user.Name, user.Age — заполнены
    c.JSON(201, user)
})

gin.H — удобный способ создать JSON:

c.JSON(200, gin.H{
    "status":  "ok",
    "message": "Успешно",
    "data":    user,
})
Middleware — встроенные и кастомные

Middleware в Gin — это функции, которые выполняются до или после обработчика.

Встроенные middleware:

func main() {
    r := gin.New()
    
    // Восстановление после паники (обязательно!)
    r.Use(gin.Recovery())
    
    // Логирование запросов
    r.Use(gin.Logger())
    
    // CORS (сторонний)
    r.Use(cors.Default())
    
    r.GET("/ping", func(c *gin.Context) {
        c.String(200, "pong")
    })
    
    r.Run()
}

Кастомный middleware:

// Middleware для проверки токена
func authMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        token := c.GetHeader("Authorization")
        
        if token == "" || token != "secret-token" {
            c.AbortWithStatusJSON(401, gin.H{
                "error": "Не авторизован",
            })
            return
        }
        
        // Передаём дальше
        c.Next()
    }
}

// Использование
func main() {
    r := gin.Default()
    
    // Для всех маршрутов
    r.Use(authMiddleware())
    
    // Только для конкретного маршрута
    r.GET("/admin", authMiddleware(), adminHandler)
}

Middleware для логирования времени:

func loggerMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()
        
        // Передаём дальше
        c.Next()
        
        // После выполнения
        duration := time.Since(start)
        log.Printf("%s %s — %v", c.Request.Method, c.Request.URL.Path, duration)
    }
}
Группы маршрутов (Route Groups)

Группы позволяют объединять маршруты с общими настройками.

func main() {
    r := gin.Default()
    
    // Группа API v1
    v1 := r.Group("/api/v1")
    {
        v1.GET("/users", getUsers)
        v1.GET("/users/:id", getUser)
        v1.POST("/users", createUser)
    }
    
    // Группа с middleware
    admin := r.Group("/admin")
    admin.Use(authMiddleware())
    {
        admin.GET("/stats", getStats)
        admin.POST("/settings", updateSettings)
    }
    
    // Вложенные группы
    api := r.Group("/api")
    {
        public := api.Group("/public")
        {
            public.GET("/ping", ping)
        }
        
        private := api.Group("/private")
        private.Use(authMiddleware())
        {
            private.GET("/profile", profile)
        }
    }
    
    r.Run()
}
Валидация данных — binding и теги

Gin использует пакет go-playground/validator для валидации.

Теги валидации:

type User struct {
    ID    int    `json:"id"`
    Name  string `json:"name" binding:"required,min=2,max=50"`
    Email string `json:"email" binding:"required,email"`
    Age   int    `json:"age" binding:"gte=0,lte=150"`
    Phone string `json:"phone" binding:"omitempty,phone"`
}

Использование:

r.POST("/users", func(c *gin.Context) {
    var user User
    
    if err := c.ShouldBindJSON(&user); err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }
    
    // Если дошли сюда — данные валидны
    c.JSON(201, user)
})

Кастомная валидация:

func customValidator() gin.HandlerFunc {
    return func(c *gin.Context) {
        var user User
        c.ShouldBindJSON(&user)
        
        if user.Age < 18 && user.Name == "" {
            c.JSON(400, gin.H{"error": "Для несовершеннолетних имя обязательно"})
            c.Abort()
            return
        }
        
        c.Next()
    }
}
Обработка ошибок и статус-коды

Gin предоставляет удобные методы для отправки ошибок.

func getUserHandler(c *gin.Context) {
    id := c.Param("id")
    
    user, err := getUserByID(id)
    if err != nil {
        // 404 Not Found
        c.JSON(404, gin.H{
            "error": "Пользователь не найден",
        })
        return
    }
    
    c.JSON(200, user)
}

Стандартные HTTP-коды:

  • c.JSON(200, data) — OK
  • c.JSON(201, data) — Created
  • c.JSON(400, gin.H{"error": "..."}) — Bad Request
  • c.JSON(401, gin.H{"error": "..."}) — Unauthorized
  • c.JSON(403, gin.H{"error": "..."}) — Forbidden
  • c.JSON(404, gin.H{"error": "..."}) — Not Found
  • c.JSON(500, gin.H{"error": "..."}) — Internal Server Error

Аборт (прерывание) с ошибкой:

func checkAccess(c *gin.Context) {
    if !isAdmin(c) {
        c.AbortWithStatusJSON(403, gin.H{
            "error": "Доступ запрещён",
        })
        return
    }
    c.Next()
}
Gin vs net/http vs Echo — сравнение
Характеристикаnet/httpGinEcho
Скорость Очень быстрый Быстрый Очень быстрый
Простота Самая простая Средняя Средняя
Роутинг Базовый (с 1.22 — хороший) Отличный Отличный
Middleware Свои Много встроенных Много встроенных
JSON encoding/json Встроенный + binding Встроенный + binding
Популярность Стандарт 🔥 Самый популярный Популярный

Когда что выбрать:

  • net/http — для простых проектов, когда не нужен фреймворк
  • Gin — для REST API, микросервисов (стандарт индустрии)
  • Echo — когда нужна чуть большая производительность
  • Fiber — когда нужна максимальная производительность (фаст-http)
Работа с файлами — загрузка и отдача

Gin поддерживает загрузку и отправку файлов.

Загрузка файла:

r.POST("/upload", func(c *gin.Context) {
    // Получаем файл из формы
    file, err := c.FormFile("file")
    if err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }
    
    // Сохраняем на диск
    err = c.SaveUploadedFile(file, "uploads/"+file.Filename)
    if err != nil {
        c.JSON(500, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(200, gin.H{
        "message": "Файл загружен",
        "filename": file.Filename,
        "size": file.Size,
    })
})

Отдача файла:

r.GET("/file/:name", func(c *gin.Context) {
    name := c.Param("name")
    c.File("uploads/" + name)
})

// Отдача как вложение (скачивание)
r.GET("/download/:name", func(c *gin.Context) {
    name := c.Param("name")
    c.FileAttachment("uploads/"+name, name)
})
Конфигурация сервера — порт, таймауты

Стандартный запуск:

r := gin.Default()
r.Run()  // :8080
r.Run(":3000")  // :3000

С настройками через http.Server:

func main() {
    r := gin.Default()
    
    srv := &http.Server{
        Addr:           ":8080",
        Handler:        r,
        ReadTimeout:    5 * time.Second,
        WriteTimeout:   10 * time.Second,
        MaxHeaderBytes: 1 << 20,  // 1 МБ
    }
    
    if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
        log.Fatal(err)
    }
}

Graceful shutdown (корректное завершение):

func main() {
    r := gin.Default()
    
    srv := &http.Server{
        Addr:    ":8080",
        Handler: r,
    }
    
    go func() {
        if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
            log.Fatal(err)
        }
    }()
    
    // Ждём сигнал
    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit
    
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    
    if err := srv.Shutdown(ctx); err != nil {
        log.Fatal("Ошибка завершения:", err)
    }
}
Тестирование Gin-приложений

Gin имеет встроенную поддержку тестирования через httptest.

func TestPingHandler(t *testing.T) {
    // Создаём тестовый роутер
    r := gin.Default()
    r.GET("/ping", func(c *gin.Context) {
        c.JSON(200, gin.H{"message": "pong"})
    })
    
    // Создаём запрос
    req, err := http.NewRequest("GET", "/ping", nil)
    if err != nil {
        t.Fatal(err)
    }
    
    // Записываем ответ
    w := httptest.NewRecorder()
    r.ServeHTTP(w, req)
    
    // Проверяем статус
    if w.Code != http.StatusOK {
        t.Errorf("Ожидался 200, получили %d", w.Code)
    }
    
    // Проверяем JSON
    var response map[string]string
    err = json.Unmarshal(w.Body.Bytes(), &response)
    if err != nil {
        t.Fatal(err)
    }
    
    if response["message"] != "pong" {
        t.Errorf("Ожидалось 'pong', получили %s", response["message"])
    }
}

Для тестирования с middleware:

func TestAuthMiddleware(t *testing.T) {
    r := gin.Default()
    r.Use(authMiddleware())
    r.GET("/protected", protectedHandler)
    
    req, _ := http.NewRequest("GET", "/protected", nil)
    req.Header.Set("Authorization", "secret-token")
    
    w := httptest.NewRecorder()
    r.ServeHTTP(w, req)
    
    if w.Code != http.StatusOK {
        t.Errorf("Ожидался 200, получили %d", w.Code)
    }
}
Gin + GORM — пример REST API

Gin часто используют вместе с GORM (ORM) для создания полноценных API.

package main

import (
    "github.com/gin-gonic/gin"
    "gorm.io/driver/sqlite"
    "gorm.io/gorm"
)

type User struct {
    ID   uint   `json:"id"`
    Name string `json:"name"`
    Age  int    `json:"age"`
}

func main() {
    // Подключаем БД
    db, _ := gorm.Open(sqlite.Open("test.db"), &gorm.Config{})
    db.AutoMigrate(&User{})
    
    r := gin.Default()
    
    // CREATE
    r.POST("/users", func(c *gin.Context) {
        var user User
        c.ShouldBindJSON(&user)
        db.Create(&user)
        c.JSON(201, user)
    })
    
    // READ — все
    r.GET("/users", func(c *gin.Context) {
        var users []User
        db.Find(&users)
        c.JSON(200, users)
    })
    
    // READ — один
    r.GET("/users/:id", func(c *gin.Context) {
        id := c.Param("id")
        var user User
        if err := db.First(&user, id).Error; err != nil {
            c.JSON(404, gin.H{"error": "Не найден"})
            return
        }
        c.JSON(200, user)
    })
    
    // UPDATE
    r.PUT("/users/:id", func(c *gin.Context) {
        id := c.Param("id")
        var user User
        if err := db.First(&user, id).Error; err != nil {
            c.JSON(404, gin.H{"error": "Не найден"})
            return
        }
        c.ShouldBindJSON(&user)
        db.Save(&user)
        c.JSON(200, user)
    })
    
    // DELETE
    r.DELETE("/users/:id", func(c *gin.Context) {
        id := c.Param("id")
        db.Delete(&User{}, id)
        c.JSON(204, nil)
    })
    
    r.Run()
}
