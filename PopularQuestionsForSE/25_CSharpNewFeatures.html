C# 9: record, init, with.
обяз

record — ссылочный тип с value-семантикой.

// record с позиционным синтаксисом
public record Person(string Name, int Age);

// record с явным объявлением
public record Person
{
    public string Name { get; init; }
    public int Age { get; init; }
}

// Использование
var p1 = new Person("Alice", 30);
var p2 = p1 with { Age = 31 };  // Копирование с изменением

// Сравнение по значению
var p3 = new Person("Alice", 30);
Console.WriteLine(p1 == p3);  // True

init — свойство, которое можно установить только при создании.

public class Person
{
    public string Name { get; init; }
    public int Age { get; init; }
}

// Можно при создании
var p = new Person { Name = "Alice", Age = 30 };
// Нельзя после создания
// p.Name = "Bob"; // Ошибка

with — выражение для копирования record с изменением.

var p2 = p1 with { Age = 31 };
C# 10: struct-записи, file-модификатор.
mid

record struct — значимый тип с value-семантикой.

public record struct Point(int X, int Y);

// Отличие от record class
// record struct — хранится в стеке (как struct)
// record class — хранится в куче (как class)

file — модификатор доступа, ограничивающий видимость в пределах файла.

// file class — виден только в этом файле
file class InternalHelper
{
    public void DoWork() { }
}

// Не виден из других файлов

Улучшения лямбд:

// Natural type для лямбд
var add = (int a, int b) => a + b;  // Func<int, int, int>

// Атрибуты для лямбд
var func = [MyAttribute] (int x) => x * 2;

Const interpolated strings

const string Prefix = "Hello";
const string Full = $"{Prefix} World";  // C# 10
C# 11: required, raw string literals.
mid

required — обязательное свойство при создании объекта.

public class Person
{
    public required string Name { get; init; }
    public required int Age { get; init; }
    public string? Email { get; init; }
}

// Обязательно указать Name и Age
var p = new Person { Name = "Alice", Age = 30 };

Raw string literals — многострочные строки без экранирования.

string json = """
{
    "name": "Alice",
    "age": 30
}
""";

// С отступами
string sql = """
    SELECT *
    FROM Users
    WHERE Age > 18
    """;

static abstract в интерфейсах

public interface IMath<T>
{
    static abstract T Add(T a, T b);
    static abstract T Zero { get; }
}

public struct IntMath : IMath<int>
{
    public static int Add(int a, int b) => a + b;
    public static int Zero => 0;
}
C# 12: primary constructors, collection expressions.
mid

Primary constructors — конструктор в объявлении класса.

// До C# 12
public class User
{
    private readonly string _name;
    public User(string name) => _name = name;
}

// C# 12
public class User(string name)
{
    public string Name => name;
}

Collection expressions — новый синтаксис для создания коллекций.

// До C# 12
int[] arr1 = new int[] { 1, 2, 3 };
int[] arr2 = { 1, 2, 3 };
List<int> list = new List<int> { 1, 2, 3 };

// C# 12
int[] arr = [1, 2, 3];
List<int> list = [1, 2, 3];
Span<int> span = [1, 2, 3];

// Spread оператор
int[] more = [..arr, 4, 5];  // [1, 2, 3, 4, 5]

alias any type

using Point = (int X, int Y);
using IntList = List<int>;

Point p = (10, 20);
IntList numbers = [1, 2, 3];
